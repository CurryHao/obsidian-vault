# 基于时间维度的复杂并发变更策略

## 基于时间维度的复杂并发变更策略

在真实生产环境中，负载绝非简单的数学曲线，而是**多阶段业务状态切换**、**外部时序数据回放**、**日历感知**以及**闭环自适应调整**的综合体。本节将深入 Locust `LoadTestShape` 的高级用法，实现**完全可编程的时序并发控制**，覆盖企业级容量规划的所有复杂场景。

---

### 策略一：有限状态机（FSM）多阶段调度

超越简单的“阶梯+保持”，引入**业务阶段语义**（预热 → 稳态 → 混合场景 → 熔断测试 → 恢复）。每个阶段可独立配置：用户数、持续时间、爬坡策略、甚至预期行为断言。

```python
# fsm_stage_shape.py
from locust import LoadTestShape
from enum import Enum
from dataclasses import dataclass
from typing import Optional
import time
import math

class Stage(Enum):
    WARMUP = "warmup"          # 预热，低并发建立连接池
    STEADY = "steady"          # 稳态，验证基线性能
    MIXED_BURST = "burst"      # 混合突发，加入写操作
    STRESS = "stress"          # 极限压力，寻找拐点
    COOLDOWN = "cooldown"      # 冷却，观察系统恢复能力
    END = "end"

@dataclass
class StageConfig:
    name: Stage
    target_users: int
    duration: int              # 秒
    ramp_duration: int         # 爬坡秒数（0表示瞬间切换）
    spawn_rate_limit: int      # 最大启动速率

class FSMShape(LoadTestShape):
    # 定义阶段时间线（按顺序执行）
    STAGES = [
        StageConfig(Stage.WARMUP, 50, 60, 30, 10),
        StageConfig(Stage.STEADY, 500, 180, 60, 50),
        StageConfig(Stage.MIXED_BURST, 1200, 120, 30, 100),
        StageConfig(Stage.STRESS, 3000, 180, 90, 150),
        StageConfig(Stage.COOLDOWN, 100, 120, 60, 20),
        StageConfig(Stage.END, 0, 1, 0, 0),  # 终止标记
    ]

    def __init__(self):
        super().__init__()
        self.stage_start_time = 0
        self.current_stage_idx = 0
        self.last_target = 0

    def tick(self):
        run_time = self.get_run_time()
        
        # 如果刚启动或阶段切换
        if run_time == 0:
            self.stage_start_time = 0
            self.current_stage_idx = 0

        # 获取当前阶段配置
        if self.current_stage_idx >= len(self.STAGES):
            return None
        
        stage = self.STAGES[self.current_stage_idx]
        if stage.name == Stage.END:
            return None

        # 计算在当前阶段的相对时间
        elapsed_in_stage = run_time - self.stage_start_time

        # 阶段结束，切换到下一个
        if elapsed_in_stage >= stage.duration:
            self.current_stage_idx += 1
            self.stage_start_time = run_time
            # 递归调用本 tick（下一轮生效，但为了即时响应，直接返回新阶段起点）
            if self.current_stage_idx < len(self.STAGES):
                next_stage = self.STAGES[self.current_stage_idx]
                # 在下一个 tick 中处理新阶段
                return self._calc_target(run_time, next_stage)
            return None

        return self._calc_target(run_time, stage)

    def _calc_target(self, run_time, stage: StageConfig):
        elapsed = run_time - self.stage_start_time
        
        # 爬坡逻辑
        if elapsed < stage.ramp_duration:
            progress = elapsed / stage.ramp_duration
            # 使用S型曲线（smoothstep）让爬坡更平滑
            smooth = progress * progress * (3 - 2 * progress)
            # 如果前一阶段有遗留用户，从上一阶段目标开始爬升
            if self.current_stage_idx > 0:
                prev_stage = self.STAGES[self.current_stage_idx - 1]
                prev_target = prev_stage.target_users
                target = int(prev_target + (stage.target_users - prev_target) * smooth)
            else:
                target = int(stage.target_users * smooth)
        else:
            target = stage.target_users
        
        # 动态 spawn_rate：剩余时间足够则平滑，否则加快
        remaining = stage.duration - elapsed
        if remaining < stage.ramp_duration:
            spawn_rate = max(10, abs(target - self.last_target) / 5)
        else:
            spawn_rate = min(stage.spawn_rate_limit, max(5, abs(target - self.last_target) / 10))
        
        self.last_target = target
        return target, int(spawn_rate) + 1
```

### 策略二：外部时间表回放（CSV/时序数据库驱动）

从生产监控系统（如 Prometheus）导出历史并发用户数（或 QPS），转换为时间序列 CSV，让 Locust **精确回放**过去的流量模式，用于回归验证或容量复盘。

```python
# timeseries_replay_shape.py
from locust import LoadTestShape
import csv
import bisect
from typing import List, Tuple

class CSVReplayShape(LoadTestShape):
    """
    从CSV加载时间序列: 格式为 "timestamp_seconds, user_count"
    例如:
        0, 100
        60, 150
        120, 300
        ...
    支持线性插值，精确回放。
    """
    def __init__(self, csv_path: str = "traffic_history.csv"):
        super().__init__()
        self.schedule: List[Tuple[float, int]] = []
        with open(csv_path, 'r') as f:
            reader = csv.reader(f)
            next(reader, None)  # 跳过表头（如果有）
            for row in reader:
                if len(row) >= 2:
                    t = float(row[0])
                    users = int(float(row[1]))
                    self.schedule.append((t, users))
        
        if not self.schedule:
            raise ValueError("时间表为空")
        
        # 确保按时间排序
        self.schedule.sort(key=lambda x: x[0])
        self.times = [s[0] for s in self.schedule]
        self.users = [s[1] for s in self.schedule]
        self.total_duration = self.times[-1]

    def tick(self):
        run_time = self.get_run_time()
        
        if run_time > self.total_duration:
            return None  # 回放结束
        
        # 二分查找当前区间
        idx = bisect.bisect_right(self.times, run_time) - 1
        idx = max(0, min(idx, len(self.times) - 2))
        
        t0, u0 = self.times[idx], self.users[idx]
        t1, u1 = self.times[idx+1], self.users[idx+1]
        
        if t1 == t0:
            target = u0
        else:
            progress = (run_time - t0) / (t1 - t0)
            target = int(u0 + (u1 - u0) * progress)
        
        # 自适应速率：根据回放曲线的斜率
        slope = (u1 - u0) / (t1 - t0 + 0.001)
        spawn_rate = max(5, int(abs(slope) * 2) + 1)
        
        return target, spawn_rate
```

### 策略三：日历感知（工作日/节假日/促销日）

很多业务负载与**星期几**强相关。通过 Python `datetime` 计算当前是工作日还是周末，动态调整负载基线。

```python
# calendar_aware_shape.py
from locust import LoadTestShape
from datetime import datetime, timedelta
import math

class CalendarAwareShape(LoadTestShape):
    """
    根据运行时的真实日期时间（非运行时长）决定负载。
    注意：这会导致每次运行负载模式固定于当前日期，适合长期巡检。
    """
    # 工作日负载峰值 2000，周末峰值 800
    weekday_peak = 2000
    weekend_peak = 800
    
    # 早高峰 8:00-10:00，晚高峰 18:00-21:00
    morning_peak_start = 8 * 3600
    morning_peak_end = 10 * 3600
    evening_peak_start = 18 * 3600
    evening_peak_end = 21 * 3600
    
    def tick(self):
        # 获取当前真实时间（而非测试运行时长）
        now = datetime.now()
        # 转换为当天的秒数
        seconds_since_midnight = (now - now.replace(hour=0, minute=0, second=0, microsecond=0)).total_seconds()
        
        # 判断是否周末（周六=5，周日=6，取决于 datetime.weekday()）
        is_weekend = now.weekday() >= 5
        
        peak_users = self.weekend_peak if is_weekend else self.weekday_peak
        
        # 计算日内系数（0~1）
        if self.morning_peak_start <= seconds_since_midnight <= self.morning_peak_end:
            # 早高峰：余弦波上升
            progress = (seconds_since_midnight - self.morning_peak_start) / (self.morning_peak_end - self.morning_peak_start)
            factor = math.sin(progress * math.pi)  # 0->1->0
        elif self.evening_peak_start <= seconds_since_midnight <= self.evening_peak_end:
            progress = (seconds_since_midnight - self.evening_peak_start) / (self.evening_peak_end - self.evening_peak_start)
            factor = math.sin(progress * math.pi)
        else:
            # 非高峰：低基线 0.2~0.4
            factor = 0.3 + 0.1 * math.sin(seconds_since_midnight / 3600)  # 微小波动
        
        target = int(peak_users * factor) + 50  # 最低50用户
        spawn_rate = max(5, target // 10)
        
        return target, spawn_rate
```

### 策略四：闭环自适应控制（基于时间窗口的 SLO 反馈）

这是**智能压测**的核心：根据前一个时间窗口的**实际响应时间**动态调整后续并发量，自动寻找系统最优容量点（类似 TCP 拥塞控制）。

```python
# adaptive_closed_loop_shape.py
from locust import LoadTestShape
from locust.runners import MasterRunner
import statistics

class AdaptiveShape(LoadTestShape):
    """
    每30秒评估一次过去窗口的平均响应时间：
    - 若 RT < 200ms，则增加 10% 用户（加性增长）
    - 若 RT 在 200-500ms，维持不变
    - 若 RT > 500ms，则降低 20% 用户（乘性减少）
    """
    window_size = 30          # 评估窗口（秒）
    min_users = 50
    max_users = 5000
    current_users = 100
    rt_threshold_low = 200    # ms
    rt_threshold_high = 500   # ms
    
    def __init__(self):
        super().__init__()
        self.last_adjust_time = 0

    def tick(self):
        run_time = self.get_run_time()
        
        # 每隔 window_size 秒调整一次
        if run_time - self.last_adjust_time < self.window_size:
            # 维持当前用户数，但需要输出给 runner
            return self.current_users, 10
        
        # 收集上一窗口的统计（仅在 Master 有效）
        if self.runner and hasattr(self.runner, 'stats'):
            # 获取所有请求的聚合统计
            total_stats = self.runner.stats.total
            if total_stats.num_requests > 0:
                avg_rt = total_stats.avg_response_time
                
                # 自适应算法
                if avg_rt < self.rt_threshold_low:
                    # 性能良好，线性增加（加法增长）
                    increment = max(10, int(self.current_users * 0.1))
                    self.current_users = min(self.max_users, self.current_users + increment)
                    action = f"↑ +{increment}"
                elif avg_rt > self.rt_threshold_high:
                    # 性能恶化，快速减少（乘法减少）
                    decrement = int(self.current_users * 0.2)
                    self.current_users = max(self.min_users, self.current_users - decrement)
                    action = f"↓ -{decrement}"
                else:
                    action = "→ 稳定"
                
                print(f"[{run_time:.0f}s] AvgRT={avg_rt:.0f}ms, Users={self.current_users}, Action={action}")
        
        self.last_adjust_time = run_time
        
        # 返回目标用户数和合理的 spawn_rate
        spawn_rate = max(5, abs(self.current_users - (self.runner.user_count if self.runner else 0)) // 5)
        return self.current_users, int(spawn_rate) + 1
```

> **⚠️ 分布式注意**：`self.runner.stats` 在 Master 上可用，Worker 不可用。此形状必须运行在 Master 节点。

### 策略五：多租户独立时间线（同脚本不同用户群）

某些场景需要**不同用户群**（如 VIP 用户、普通用户、游客）在不同时间开始压测，模拟差异化 SLA。

```python
# multi_tenant_shape.py
from locust import LoadTestShape

class MultiTenantShape(LoadTestShape):
    """
    定义多个用户群的时间线，返回总用户数。
    但需配合自定义 User 类中的 `user_type` 和 `weight` 动态分配。
    """
    def tick(self):
        t = self.get_run_time()
        
        # 租户A（普通用户）：0秒开始，2000用户
        tenant_a = 2000 if t > 0 else 0
        # 租户B（VIP用户）：延迟30秒启动，500用户
        tenant_b = 500 if t > 30 else 0
        # 租户C（游客）：延迟120秒启动，3000用户（模拟突发）
        tenant_c = 3000 if 120 <= t < 240 else 0
        
        total_users = tenant_a + tenant_b + tenant_c
        spawn_rate = max(10, total_users // 10)
        
        return total_users, spawn_rate
```

在对应的 `User` 类中，通过 `self.user_id % 3` 或全局标志分配不同的 `task_set`，实现**业务行为与时间线联动**。

### 六、高级综合实战：动态任务权重随阶段切换

在某些阶段，我们希望**只压测读接口**（预热），后续阶段加入**写接口**（混合）。单纯的用户数控制无法实现，需要在 `User` 类中引入**全局阶段变量**。

```python
# 全局阶段标志（在 LoadTestShape 中更新）
from locust import events
import json

current_phase = "warmup"  # 由 Shape 更新

@events.init.add_listener
def on_init(environment, **kwargs):
    environment.current_phase = "warmup"

class PhaseAwareUser(HttpUser):
    @task
    def read_only(self):
        if self.environment.current_phase in ["warmup", "steady"]:
            self.client.get("/api/read")
    
    @task
    def write_operation(self):
        if self.environment.current_phase in ["burst", "stress"]:
            self.client.post("/api/write", json={"data": "test"})
```

在 `FSMShape` 的 `tick()` 中，阶段切换时同步更新 `self.environment.current_phase`。

### 七、执行与监控集成最佳实践

| 策略类型 | 推荐执行参数 | 关键观测指标 |
|---------|------------|------------|
| **FSM状态机** | `--csv=fsm` + `--html=fsm.html` | 阶段时间线与吞吐量叠加图 |
| **CSV回放** | `--loglevel INFO` 观察回放进度 | 实际用户数 vs 期望用户数（CSV输出对比） |
| **日历感知** | 结合Cron定时运行（如每周一8点） | 工作日 vs 周末失败率差异 |
| **闭环自适应** | 必须单机或分布式Master，`--web-port` 可视化 | 响应时间与用户数变化曲线，观察收敛性 |
| **多租户** | 配合 `--tags` 或 `--exclude-tags` | 各租户独立统计（需自定义 `name`） |

### 八、总结与选型建议

| 业务场景 | 推荐策略 |
|---------|---------|
| **容量规划/拐点测试** | 策略一：FSM 状态机（阶梯+保持） |
| **生产流量回放/复盘** | 策略二：CSV 时间表回放 |
| **7x24小时巡检（区分节假日）** | 策略三：日历感知 |
| **弹性伸缩策略验证（HPA）** | 策略四：闭环自适应（模拟自动扩缩容） |
| **混合业务优先级隔离** | 策略五：多租户时间线 |

**组合使用**：你可以将**日历感知**与**FSM状态机**结合——工作日执行高负载 FSM，周末执行低负载 FSM。只需在 `__init__` 中根据 `datetime.now()` 动态选择 `STAGES` 配置。

如果你有特定的生产监控数据（如 Prometheus 导出的历史 QPS 曲线），或需要实现**“基于响应时间的 PID 控制”**（比简单的加减法更平滑），欢迎提供数据格式，我可以定制对应的 `LoadTestShape` 实现。
